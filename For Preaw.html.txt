<script>
// --- WEBHOOK & NOTIFICATION LOGIC ---

// Show Toast popup message
function showToast(message, icon = "✨") {
const toast = document.getElementById('toast');
document.getElementById('toast-text').innerText = message;
document.getElementById('toast-icon').innerText = icon;
toast.classList.remove('opacity-0', 'pointer-events-none');
toast.classList.add('opacity-100');
setTimeout(() => {
toast.classList.remove('opacity-100');
toast.classList.add('opacity-0', 'pointer-events-none');
}, 3500);
}

// Settings Modal Controllers
function openSettingsModal() {
loadSavedSettings();
const modal = document.getElementById('settings-modal');
modal.classList.remove('hidden');
setTimeout(() => modal.classList.remove('opacity-0'), 10);
}

function closeSettingsModal() {
const modal = document.getElementById('settings-modal');
modal.classList.add('opacity-0');
setTimeout(() => modal.classList.add('hidden'), 300);
}

function toggleWebhookFields() {
const type = document.getElementById('webhook-type').value;
document.getElementById('field-discord').classList.toggle('hidden', type !== 'discord');
document.getElementById('field-telegram').classList.toggle('hidden', type !== 'telegram');
document.getElementById('field-custom').classList.toggle('hidden', type !== 'custom');
}

function saveSettings() {
const type = document.getElementById('webhook-type').value;
localStorage.setItem('fluk_webhook_type', type);
localStorage.setItem('fluk_discord_url', document.getElementById('discord-url').value.trim());
localStorage.setItem('fluk_telegram_token', document.getElementById('telegram-token').value.trim());
localStorage.setItem('fluk_telegram_chatid', document.getElementById('telegram-chatid').value.trim());
localStorage.setItem('fluk_custom_url', document.getElementById('custom-url').value.trim());

showToast("บันทึกการตั้งค่าการแจ้งเตือนแล้ว! 💾", "✅");
closeSettingsModal();
}

function loadSavedSettings() {
const type = localStorage.getItem('fluk_webhook_type') || 'discord';
document.getElementById('webhook-type').value = type;
document.getElementById('discord-url').value = localStorage.getItem('fluk_discord_url') || '';
document.getElementById('telegram-token').value = localStorage.getItem('fluk_telegram_token') || '';
document.getElementById('telegram-chatid').value = localStorage.getItem('fluk_telegram_chatid') || '';
document.getElementById('custom-url').value = localStorage.getItem('fluk_custom_url') || '';
toggleWebhookFields();
}

// Send Notification Function (Discord / Telegram / Custom)
async function sendNotification(message, title = "เว็บไซต์สารภาพรัก 🌙", isTest = false) {
const type = localStorage.getItem('fluk_webhook_type') || 'discord';
const timestamp = new Date().toLocaleString('th-TH', { timeZone: 'Asia/Bangkok' });

try {
if (type === 'discord') {
const url = localStorage.getItem('fluk_discord_url');
if (!url) return false;

const payload = {
username: "หัวใจของฟลุ๊ค 💖",
avatar_url: "https://cdn-icons-png.flaticon.com/512/2904/2904973.png",
embeds: [{
title: title,
description: message,
color: isTest ? 3447003 : 16007990, // Blue if test, Pink if real
footer: { text: `เวลาส่ง: ${timestamp}` }
}]
};

await fetch(url, {
method: 'POST',
headers: { 'Content-Type': 'application/json' },
body: JSON.stringify(payload)
});
return true;

} else if (type === 'telegram') {
const token = localStorage.getItem('fluk_telegram_token');
const chatId = localStorage.getItem('fluk_telegram_chatid');
if (!token || !chatId) return false;

const telegramUrl = `https://api.telegram.org/bot${token}/sendMessage`;
const payload = {
chat_id: chatId,
text: `*${title}*\n\n${message}\n\n_เวลา: ${timestamp}_`,
parse_mode: "Markdown"
};

await fetch(telegramUrl, {
method: 'POST',
headers: { 'Content-Type': 'application/json' },
body: JSON.stringify(payload)
});
return true;

} else if (type === 'custom') {
const url = localStorage.getItem('fluk_custom_url');
if (!url) return false;

await fetch(url, {
method: 'POST',
headers: { 'Content-Type': 'application/json' },
body: JSON.stringify({ title, message, timestamp, isTest })
});
return true;
}
} catch (err) {
console.error("Webhook notification error:", err);
return false;
}
return false;
}

// --- AUDIO ENGINE ---
let audioCtx = null;
let windNoise = null;
let isAudioPlaying = false;

function toggleSound() {
if (!isAudioPlaying) {
try {
audioCtx = new (window.AudioContext || window.webkitAudioContext)();
const bufferSize = audioCtx.sampleRate * 2;
const noiseBuffer = audioCtx.createBuffer(1, bufferSize, audioCtx.sampleRate);
const output = noiseBuffer.getChannelData(0);
for (let i = 0; i < bufferSize; i++) {
output[i] = Math.random() * 2 - 1;
}

windNoise = audioCtx.createBufferSource();
windNoise.buffer = noiseBuffer;
windNoise.loop = true;

const filter = audioCtx.createBiquadFilter();
filter.type = 'lowpass';
filter.frequency.value = 350;

const gainNode = audioCtx.createGain();
gainNode.gain.value = 0.15;

windNoise.connect(filter);
filter.connect(gainNode);
gainNode.connect(audioCtx.destination);

windNoise.start();
isAudioPlaying = true;
document.getElementById('sound-icon').innerText = '🔊';
document.getElementById('sound-text').innerText = 'ปิดเสียงลม';
} catch(e) {
console.log('Audio error:', e);
}
} else {
if (windNoise) { windNoise.stop(); windNoise.disconnect(); }
if (audioCtx) { audioCtx.close(); }
isAudioPlaying = false;
document.getElementById('sound-icon').innerText = 'hh';
document.getElementById('sound-icon').innerText = '🔇';
document.getElementById('sound-text').innerText = 'เปิดเสียงลม';
}
}
